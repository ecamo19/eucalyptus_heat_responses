2. Calculate the relative water content

euc_sideroxylum_rwc_data <- 
  
    euc_sideroxylum_droughtbox_data_cleaned %>%

        group_by(string_number, set_temperature, vpd_control) %>%
        
        # Calculate relative water content (%)
        mutate(relative_water_content_perc = 
                 (string_weight_grams - sample_dry_weight_g) / 
                 (max(string_weight_grams) - sample_dry_weight_g),
               
               # Hook weight, prensa weight have been already added when
               # creating sample_dry_weight_g
               branch_dry_matter_content_g_g =
                    sample_dry_weight_g/(max(string_weight_grams))
               
               )
